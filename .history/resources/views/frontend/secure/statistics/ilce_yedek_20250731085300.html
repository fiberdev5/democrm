@endsection public function IlceStatistics($tenant_id) { return view('frontend.secure.statistics.ilce_statistics', compact('tenant_id')); } public function getIlceStatisticsData(Request $request, $tenant_id) { if ($request->ajax()) { $from_date = Carbon::parse($request->input('from_date'))->startOfDay(); $to_date = Carbon::parse($request->input('to_date'))->endOfDay(); $query = DB::table('services as s') ->join('customers as c', 's.musteri_id', '=', 'c.id') ->join('ilces as i', 'c.ilce', '=', 'i.id') ->select( 'i.ilceName as ilce', DB::raw('COUNT(s.id) as toplam'), 'i.id as ilce_id' ) ->where('s.firma_id', $tenant_id) ->whereBetween('s.created_at', [$from_date, $to_date]) // bu tarih filtresini unutma ->groupBy('i.ilceName', 'i.id'); // Arama if ($request->has('search') && !empty($request->search['value'])) { $searchValue = $request->search['value']; $query->where('i.ilceName', 'LIKE', "%$searchValue%"); } // Sıralama if ($request->has('order')) { $orderColumn = $request->order[0]['column']; $orderDirection = $request->order[0]['dir']; $columns = ['ilce', 'toplam']; if (isset($columns[$orderColumn])) { $query->orderBy($columns[$orderColumn], $orderDirection); } } else { $query->orderByDesc('toplam'); } // Filtreli kayıt sayısı $filteredRecords = $query->count(); // Sayfalama if ($request->has('start') && $request->has('length')) { $query->offset($request->start)->limit($request->length); } // En son çalıştır $data = $query->get(); // Tüm kayıt sayısı $totalRecords = DB::table('services as s') ->join('customers as c', 's.musteri_id', '=', 'c.id') ->join('ilces as i', 'c.ilce', '=', 'i.id') ->where('s.firma_id', $tenant_id) ->distinct('i.id') ->count('i.id'); return response()->json([ 'draw' => intval($request->draw), 'recordsTotal' => $totalRecords, 'recordsFiltered' => $filteredRecords, 'data' => $data ]); } }
